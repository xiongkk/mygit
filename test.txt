
Option Explicit

' ==========================================
' 使用前请务必在 VBA编辑器中执行以下操作：
' 1. 点击菜单栏 "工具" (Tools) -> "引用" (References)
' 2. 勾选 "Microsoft HTML Object Library"
' 3. 勾选 "Microsoft ActiveX Data Objects x.x Library" (用于处理UTF-8编码，推荐6.1或2.8)
' ==========================================

Sub ImportRedmineBoard()
    Dim filePath As String
    Dim htmlContent As String
    Dim doc As New HTMLDocument
    Dim tableElem As Object
    Dim rows As Object
    Dim row As Object
    Dim ws As Worksheet
    Dim outputRow As Long
    Dim i As Long
    
    ' === 配置区域 ===
    ' 请修改此处为你导出的HTML文件完整路径
    filePath = "C:\Users\YourName\Desktop\redmine_export.html"
    ' =================
    
    ' 初始化工作表
    Set ws = ThisWorkbook.Sheets(1)
    ws.Cells.Clear
    ws.Range("A1:D1").Value = Array("Subject", "Created On", "Author", "Last Message Link")
    outputRow = 2
    
    ' 1. 读取HTML文件 (使用ADODB处理UTF-8编码，避免中文乱码)
    htmlContent = ReadUTF8File(filePath)
    If Len(htmlContent) = 0 Then
        MsgBox "文件读取失败或为空，请检查路径及编码。", vbCritical
        Exit Sub
    End If
    
    ' 2. 加载HTML到DOM
    On Error Resume Next
    doc.body.innerHTML = htmlContent
    On Error GoTo 0
    
    If doc.body.innerHTML = "" Then
        MsgBox "HTML解析失败，请确认已引用 Microsoft HTML Object Library。", vbCritical
        Exit Sub
    End If
    
    ' 3. 定位目标表格 class="list messages"
    Set tableElem = doc.getElementsByClassName("list messages")(0)
    If tableElem Is Nothing Then
        MsgBox "未在HTML中找到 class='list messages' 的表格。", vbExclamation
        Exit Sub
    End If
    
    ' 4. 遍历表格行
    Set rows = tableElem.getElementsByTagName("tr")
    
    Application.ScreenUpdating = False ' 关闭屏幕刷新以提高速度
    
    For i = 0 To rows.Length - 1
        Set row = rows(i)
        
        ' 跳过表头或空行 (通常表头是th，数据是td，这里简单判断td数量)
        If row.getElementsByTagName("td").Length > 0 Then
            Dim subject As String, createdOn As String, author As String, lastLink As String
            
            ' 提取数据 helper函数
            subject = GetCellTextByClass(row, "subject")
            createdOn = GetCellTextByClass(row, "created_on")
            author = GetCellTextByClass(row, "author")
            
            ' 特殊处理 last_message: 获取第一个链接的文本
            lastLink = GetFirstLinkText(row, "last_message")
            
            ' 只有当主题不为空时才写入，避免写入空白行
            If subject <> "" Then
                ws.Cells(outputRow, 1).Value = subject
                ws.Cells(outputRow, 2).Value = createdOn
                ws.Cells(outputRow, 3).Value = author
                ws.Cells(outputRow, 4).Value = lastLink
                outputRow = outputRow + 1
            End If
        End If
    Next i
    
    Application.ScreenUpdating = True
    MsgBox "导入完成！共提取 " & (outputRow - 2) & " 条数据。", vbInformation
End Sub

' ==========================================
' 辅助函数：根据Class名称获取单元格文本
' ==========================================
Function GetCellTextByClass(rowObj As Object, className As String) As String
    Dim cells As Object
    Dim cell As Object
    Dim i As Long
    
    Set cells = rowObj.getElementsByTagName("td")
    For i = 0 To cells.Length - 1
        Set cell = cells(i)
        ' InStr检查class属性是否包含目标类名
        If InStr(1, cell.className, className, vbTextCompare) > 0 Then
            GetCellTextByClass = Trim(cell.innerText)
            Exit Function
        End If
    Next i
    GetCellTextByClass = ""
End Function

' ==========================================
' 辅助函数：获取特定Class单元格内的第一个链接文本
' ==========================================
Function GetFirstLinkText(rowObj As Object, className As String) As String
    Dim cells As Object
    Dim cell As Object
    Dim links As Object
    Dim i As Long
    
    Set cells = rowObj.getElementsByTagName("td")
    For i = 0 To cells.Length - 1
        Set cell = cells(i)
        If InStr(1, cell.className, className, vbTextCompare) > 0 Then
            Set links = cell.getElementsByTagName("a")
            If links.Length > 0 Then
                ' 提取链接文本，如果需要URL可改为 links(0).href
                GetFirstLinkText = Trim(links(0).innerText)
            Else
                ' 如果没有链接，返回单元格纯文本
                GetFirstLinkText = Trim(cell.innerText)
            End If
            Exit Function
        End If
    Next i
    GetFirstLinkText = ""
End Function

' ==========================================
' 辅助函数：使用ADODB.Stream读取UTF-8文件
' ==========================================
Function ReadUTF8File(filePath As String) As String
    Dim stream As Object
    Dim content As String
    
    Set stream = CreateObject("ADODB.Stream")
    On Error GoTo ErrorHandler
    
    With stream
        .Type = 2 ' adTypeText
        .Charset = "UTF-8"
        .Open
        .LoadFromFile filePath
        content = .ReadText
        .Close
    End With
    
    ReadUTF8File = content
    Exit Function
    
ErrorHandler:
    ReadUTF8File = ""
End Function
